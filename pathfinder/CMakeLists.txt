cmake_minimum_required(VERSION 3.25)
project(rpg_pathfinder LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(pathfinder_core src/grid.cpp src/search.cpp src/api.cpp src/server.cpp)
target_include_directories(pathfinder_core PUBLIC include)
target_include_directories(pathfinder_core SYSTEM PUBLIC third_party)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pathfinder_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
find_package(Threads REQUIRED)
add_executable(pathfinder_tests test/test_search.cpp)
target_link_libraries(pathfinder_tests PRIVATE pathfinder_core)
add_test(NAME pathfinder_tests COMMAND pathfinder_tests)

add_executable(pathfinder_api_tests test/test_api.cpp)
target_link_libraries(pathfinder_api_tests PRIVATE pathfinder_core)
add_test(NAME pathfinder_api_tests COMMAND pathfinder_api_tests)

add_executable(pathfinder_server_tests test/test_server.cpp)
target_link_libraries(pathfinder_server_tests PRIVATE pathfinder_core)
add_test(NAME pathfinder_server_tests COMMAND pathfinder_server_tests)

add_executable(pathfinder_contract_tests test/test_contract.cpp)
target_link_libraries(pathfinder_contract_tests PRIVATE pathfinder_core)
target_compile_definitions(pathfinder_contract_tests PRIVATE CONTRACT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/contract/cases")
add_test(NAME pathfinder_contract_tests COMMAND pathfinder_contract_tests)

add_executable(pathfinder_work_queue_tests test/test_work_queue.cpp)
target_link_libraries(pathfinder_work_queue_tests PRIVATE pathfinder_core Threads::Threads)
add_test(NAME pathfinder_work_queue_tests COMMAND pathfinder_work_queue_tests)

# The service itself: the program that listens on a port.
set(PATHFINDER_BUILD "dev" CACHE STRING "Build identifier shown by GET /v1/info")
add_executable(pathfinder_server src/main.cpp)
include(cmake/httplib.cmake)
target_link_libraries(pathfinder_server PRIVATE pathfinder_core Threads::Threads)
target_compile_definitions(pathfinder_server PRIVATE PATHFINDER_BUILD="${PATHFINDER_BUILD}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pathfinder_server PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_compile_definitions(pathfinder_server PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(pathfinder_server PRIVATE ws2_32)
endif()
